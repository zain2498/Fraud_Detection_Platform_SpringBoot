CREATE TABLE fraud_payment_request AS
SELECT
    request_id,

    customer_account_id + 1000000 AS customer_account_id,

    CONCAT('TXN_', request_id) AS transaction_id,

    reference_type_id,

    reference_id,

    CONCAT('REF_', request_id) AS reference_number,

    'MASKED_PAYMENT_REFERENCE' AS payment_reference,

    'MASKED_VPS_TX_ID' AS vps_tx_id,

    'MASKED_SECURITY_KEY' AS security_key,

    'MASKED_AUTH_NO' AS tx_auth_no

FROM pps_eft_payment_request;

CREATE TABLE fraud_transaction_request AS
SELECT
    transaction_request_id,

    CONCAT('TXN_', transaction_request_id) AS transaction_id,

    request_type,

    reference_id,

    revenue_area_id,

    merchant_id,

    payment_type_id,

    status,

    response_code,

    response_text,

    amount,

    created_date

FROM pps_eft_transaction_request;


CREATE TABLE fraud_analysis (

    id BIGINT PRIMARY KEY AUTO_INCREMENT,

    request_id BIGINT,

    transaction_id VARCHAR(255),

    risk_score INT,

    decision VARCHAR(50),

    reasons TEXT,

    analyzed_at DATETIME
);
